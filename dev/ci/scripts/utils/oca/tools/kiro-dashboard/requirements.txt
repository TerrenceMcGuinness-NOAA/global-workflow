#     <interpreter> -m pip install --user -r tools/kiro-dashboard/requirements.txt

# (boto3 1.42.70, pandas 3.0.1, numpy 2.4.2, plotly 6.6.0, Jinja2 3.1.6) and on
# the Parallel Works host both satisfy this file.
